namespace Validation{

    inline std::string validateNotBlank(const std::string &value, const std::string &fieldName){
        if(value.empty()){
            throw std::invalid_argument(fieldName + " cannot be empty.");
        }
        return value;
    }

    template <typename T>
    inline T validateNotNegative(T value, const std::string &fieldName){
        static_assert(std::is_arithmetic_v<T>, "Template parameter must be numeric type.");
        if(value < 0){
            throw std::invalid_argument(fieldName + " cannot be negative.");
        }
        return value;
    }

    template <typename T>
    inline void validateAlessThanB(T lesserValue, T greaterValue, const std::string &firstName, const std::string &secondName){
        static_assert(std::is_arithmetic_v<T>, "Template parameters must be numeric type.");
        if(lesserValue >= greaterValue){
            throw std::invalid_argument(firstName + " must be less than " + secondName + ".");
        }
    }

    template <typename A, typename U>
    inline A validateBetween(A value, U minValue, U maxValue, const std::string &fieldName){
        static_assert(std::is_arithmetic_v<A>, "Template parameters must be numeric type.");
        static_assert(std::is_arithmetic_v<U>, "Template parameters must be numeric type.");
        if(value < minValue || value > maxValue){
            throw std::invalid_argument(fieldName + " should be in range [" + std::to_string(minValue) + "-" + std::to_string(maxValue) + "].");
        }
        return value;
    }
}

Agent::Agent(const std::string &id, const std::string &codeName, int age, int yearsOfExperience){
    this->id = Validation::validateNotBlank(id, "ID");
    this->codeName = Validation::validateNotBlank(codeName, "Code name");
    Validation::validateAlessThanB(yearsOfExperience, age, "Experience years", "Age");
    this->age = Validation::validateNotNegative(age, "Age");
    this->yearsOfExperience = Validation::validateNotNegative(yearsOfExperience, "Experience years");
}

std::string Agent::getId() const {return id;}
std::string Agent::getCodeName() const {return codeName;}
int Agent::getAge() const {return age;}
int Agent::getExperience() const {return yearsOfExperience;}
std::vector<std::shared_ptr<Mission> > Agent::getMissionsList() const{
    std::vector<std::shared_ptr<Mission> > missionList;
    if(missions.empty()) return missionList;
    for(const auto &value : std::views::values(missions)){
        missionList.push_back(value);
    }
    return missionList;
}


void Agent::setCodeName(const std::string &codeName){
    this->codeName = Validation::validateNotBlank(codeName, "Code name");
}

void Agent::setAge(int age){
    Validation::validateAlessThanB(this->yearsOfExperience, age, "Experience years", "Age");
    this->age = Validation::validateNotNegative(age, "Age");
}

void Agent::setExperience(int experience){
    Validation::validateAlessThanB(experience, this->age, "Experience years", "Age");
    this->yearsOfExperience = Validation::validateNotNegative(experience, "Experience years");
}

void Agent::addExperience(int experience){
    setExperience(std::min(this->yearsOfExperience + experience, age - 1));
}

void Agent::assignMission(std::shared_ptr<Mission> mission){
    if(!mission) throw NullObjectException("Mission");
    auto [it, inserted] = missions.try_emplace(mission->getId(), mission);
    if(!inserted){
        throw DuplicateMissionException();
    }
}

void Agent::removeMission(const std::string &id){
    if(id.empty()) throw EmptyTextEception("Mission ID");
    if(missions.erase(id) == 0){
        throw MissionNotFoundException(id);
    }
}

Mission* Agent::findMissionById(const std::string &id) const{
    if(id.empty()) throw EmptyTextEception("Mission ID");
    auto it = missions.find(id);
    if(it != missions.end()){
        return it->second.get();
    }
    std::cerr << "Mission with such ID was not found\n";
    return nullptr;
}

void Agent::completeMission(const std::string &id){
    if(id.empty()) throw EmptyTextEception("Mission ID");
    Mission* mission = findMissionById(id);
    if(mission) mission->completeMission();
}

void Agent::showMissions() const{
    if(missions.empty()) return;
    for(const auto &value : std::views::values(missions)){
        std::cout << value << "\n";
    }
}

bool Agent::hasCompleted(const std::string &id) const{
    if(id.empty()) throw EmptyTextEception("Mission ID");
    Mission* mission = findMissionById(id);
    if(mission){
        return mission->isComplete();
    }
    return false;
}

void Agent::submitReport(std::unique_ptr<IntelligenceReport> report, IntelligenceDepartment *department){
    if(!department) throw NullObjectException("Department");
    if(department->findAgentById(this->id) == nullptr) throw std::out_of_range("The agent must be a part of the department to sumbit reports.");
    department->addReport(std::move(report), getId());
}

void Agent::displayInfo() const{
    std::cout << "ID: " << id << "\nName: " << codeName << "\nAge: " << age << "\nExperience: " << yearsOfExperience << " years\n";
}

bool Agent::operator==(const Agent &other) const{
    return this->id == other.id;
}

        throw EmptyTextEception("Agent ID");
    }
    auto it = agents.find(id);

void IntelligenceDepartment::displayAllAgents() const{
    if(agents.empty()) return;
    std::cout << "\nAll " << agents.size() << " agents of the department!\n\n";
    for(const auto &value : std::views::values(agents)){
        value->displayInfo();
        std::cout << "\n";
    }
}

void IntelligenceDepartment::addReport(std::unique_ptr<IntelligenceReport> report, const std::string &submitterId){
    if(!report) throw NullObjectException("Report");
    if(reports.count(report->getReportId()) > 0){
        throw DuplicateReportException();        
    }
    report->setSubmitterId(submitterId);
    reports.emplace(report->getReportId(), std::move(report));
}

void IntelligenceDepartment::removeReport(const std::string &id){
    if(id.empty()) throw EmptyTextEception("Report ID");
    if(reports.erase(id) == 0){
        throw ReportNotFoundException(id);
    }
}

IntelligenceReport* IntelligenceDepartment::findReportById(const std::string &id) const{
    if(reports.empty()) return nullptr;
    auto it = reports.find(id);
    if(it != reports.end()){
        return it->second.get();
    }
    std::cerr << "Report with such ID was not found.\n";
    return nullptr;
}

void IntelligenceDepartment::displayAllReports() const{
    if(reports.empty()) return;
    std::cout << "All " << reports.size() << " reports:\n\n";
    for(const auto &value : std::views::values(reports)){
        std::cout << value << "\n";
    }
    std::cout << "\n";
}

double IntelligenceDepartment::getTotalThreatLevel() const{
    double total = 0.0; 
    for(const auto &value : std::views::values(agents)){
        total += value->calculateThreatLevel(); 
    }
    return total;
}

Agent* IntelligenceDepartment::getMostDangerousAgent() const{
    if(agents.empty()){
        return nullptr;
    }
    Agent* curr = nullptr;
    double maxThreat = -1.0;
    for(const auto &value : std::views::values(agents)){
        double currentThreat = value->calculateThreatLevel();
        if(currentThreat > maxThreat){
            maxThreat = currentThreat;
            curr = value.get();
        }
    }
    return curr;
}

std::vector<Agent*> IntelligenceDepartment::getRawAgentPointers() const{
    std::vector<Agent*> ptrs;
    ptrs.reserve(agents.size());
    for(const auto &value : std::views::values(agents)){
        ptrs.push_back(value.get());
    }
    return ptrs;
}

std::vector<Agent*> IntelligenceDepartment::sortAgentsByThreatLevel() const{
    auto sortedAgents = getRawAgentPointers();
    std::sort(sortedAgents.begin(), sortedAgents.end(), [](Agent* a, Agent* b){
        return a->calculateThreatLevel() > b->calculateThreatLevel();
    });
    return sortedAgents;
}

std::vector<Agent*> IntelligenceDepartment::getAgentsByType(const std::string &type) const{
    if(type.empty() || type != "Elite Anbu Agent" || type != "Infiltration Agent" || type != "Recon Agent"){
        throw std::invalid_argument("Please, enter a valid agent type:\n1. Elite Anbu Agent\n2. Infiltration Agent\n3. Recon Agent");
    }
    std::vector<Agent*> anbus;
    if(agents.empty()) return anbus;
    for(const auto &value : std::views::values(agents)){
        if(value->getType() == type){
            anbus.push_back(value.get());
        }
    }
    return anbus;
}

double IntelligenceDepartment::getAverageThreatLevel() const{
    if(agents.empty()) return 0.0;
    double sum = 0.0;
    for(const auto& agent : std::views::values(agents)){
        sum += static_cast<double>(agent->calculateThreatLevel());
    }
    return sum / agents.size();
}

std::tuple<int, int, int> IntelligenceDepartment::getAgentCountByType() const{
    if(agents.empty()) return {0,0,0};
    int reconCount = 0, anbuCount = 0, infiltCount = 0;
    for(const auto& agent : std::views::values(agents)){
        if(!agent) continue;
        if(dynamic_cast<reconAgent*>(agent.get())){
            reconCount++;
        }else if (dynamic_cast<eliteAnbuAgent*>(agent.get())) { // Adjust class name to match yours
            anbuCount++;
        } else if (dynamic_cast<infiltrationAgent*>(agent.get())) {
            infiltCount++;
        }
    }
    return {reconCount, anbuCount, infiltCount};
}
